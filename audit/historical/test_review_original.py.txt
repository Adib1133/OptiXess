"""Review acceptance tests. Failures identify defects, not successful injection.

All game files are temporary. Fake payloads test file orchestration only.
No DLL is loaded and no real game process is modified.
"""
import configparser
import os
from pathlib import Path
import sys
import tempfile
from types import SimpleNamespace
import unittest
from unittest.mock import Mock, patch

ROOT = Path(__file__).resolve().parents[1]
sys.path[:0] = [str(ROOT), str(ROOT / '.audit-deps')]
from core.injector import Injector
from core.safety import SafetyManager, CrashWatchdog
from core.config_generator import ConfigGenerator


class ReviewAcceptance(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory(prefix='opti_review_')
        self.addCleanup(self.tmp.cleanup)
        self.root = Path(self.tmp.name)
        self.game = self.root / 'game'
        self.game.mkdir()
        self.exe = self.game / 'review-game-unique.exe'
        self.exe.write_bytes(b'not an executable')
        self.assets = self.root / 'assets'
        self.version = self.assets / 'versions' / 'v0.9.4'
        self.version.mkdir(parents=True)
        self.payload = b'TEST-NOT-A-PE' * 10000
        for name in ['OptiScaler.dll', 'libxess.dll', 'libxess_dx11.dll', 'libxess_fg.dll', 'libxell.dll', 'fakenvapi.dll']:
            (self.version / name).write_bytes(self.payload)
        (self.version / 'fakenvapi.ini').write_text('[fakenvapi]\nforce_reflex=0\n')
        self.injector = Injector(str(self.assets))

    def apply(self, hook='dxgi.dll', **kw):
        return self.injector.apply_injection(str(self.game), str(self.exe), hook_method=hook, **kw)

    def test_all_exposed_hooks_copy_and_restore(self):
        for hook in Injector.DEFAULT_HOOKS:
            with self.subTest(hook=hook):
                path = self.game / hook
                path.write_bytes(b'original')
                result = self.apply(hook)
                self.assertTrue(result['success'], result)
                self.assertEqual(path.read_bytes(), self.payload)
                self.assertTrue(self.injector.revert_injection(str(self.game))['success'])
                self.assertEqual(path.read_bytes(), b'original')
                path.unlink()

    def test_missing_xess_must_fail_preflight(self):
        (self.version / 'libxess.dll').unlink()
        self.assertFalse(self.apply()['success'], 'Missing XeSS reported as complete suite')

    def test_legacy_proxy_source_must_be_deployed(self):
        (self.version / 'OptiScaler.dll').rename(self.version / 'dxgi.dll')
        self.assertTrue(self.injector.version_manager.is_version_valid('v0.9.4'))
        self.assertTrue(self.apply()['success'])
        self.assertTrue((self.game / 'dxgi.dll').exists(), 'Validated legacy proxy never copied')

    def test_reapply_preserves_vanilla_backup(self):
        (self.game / 'dxgi.dll').write_bytes(b'original')
        self.assertTrue(self.apply()['success'])
        self.assertTrue(self.apply()['success'])
        self.assertTrue(self.injector.revert_injection(str(self.game))['success'])
        self.assertEqual((self.game / 'dxgi.dll').read_bytes(), b'original')

    def test_fakenvapi_ini_restored(self):
        (self.game / 'fakenvapi.ini').write_text('user settings')
        self.assertTrue(self.apply()['success'])
        self.injector.revert_injection(str(self.game))
        self.assertEqual((self.game / 'fakenvapi.ini').read_text(), 'user settings')

    def test_no_manifest_preserves_native_libraries(self):
        (self.game / 'libxess.dll').write_bytes(b'native game runtime')
        self.injector.revert_injection(str(self.game))
        self.assertTrue((self.game / 'libxess.dll').exists())

    def test_restore_failure_retains_backup_and_reports_failure(self):
        (self.game / 'dxgi.dll').write_bytes(b'original')
        self.assertTrue(self.apply()['success'])
        with patch('core.safety.shutil.copy2', side_effect=PermissionError('locked')), patch('core.safety.time.sleep'):
            result = self.injector.revert_injection(str(self.game))
        self.assertEqual((result['success'], SafetyManager.has_active_backup(str(self.game))), (False, True))

    def test_backup_failure_preserves_originals(self):
        (self.game / 'dxgi.dll').write_bytes(b'original')
        with patch('core.safety.shutil.copy2', side_effect=OSError('disk full')):
            result = self.apply()
        self.assertFalse(result['success'])
        self.assertTrue((self.game / 'dxgi.dll').exists(), 'Snapshot failure deleted original')

    def test_corrupt_large_payload_rejected(self):
        self.assertFalse(self.injector.version_manager.is_version_valid('v0.9.4'))

    def test_selected_missing_version_not_silently_substituted(self):
        self.assertFalse(self.apply(optiscaler_version='v-does-not-exist')['success'])

    def test_dx11_runtime_deployed(self):
        self.assertTrue(self.apply()['success'])
        self.assertTrue((self.game / 'libxess_dx11.dll').exists())

    def test_generated_keys_match_installed_schema(self):
        official = configparser.ConfigParser(strict=False)
        official.read(ROOT / 'assets/versions/v0.9.4/OptiScaler.ini', encoding='utf-8-sig')
        generated = configparser.ConfigParser()
        generated.read_string(ConfigGenerator.generate_nvngx_ini(custom_scale=0.67))
        unknown = [f'{s}.{k}' for s in generated for k in generated[s]
                   if not official.has_option(s, k)]
        self.assertEqual(unknown, [], f'Unknown keys: {unknown}')

    def test_fg_selects_xefg_output(self):
        cfg = configparser.ConfigParser()
        cfg.read_string(ConfigGenerator.generate_nvngx_ini(frame_gen_enabled=True))
        self.assertEqual(cfg.get('FrameGen', 'FGOutput', fallback=None), 'xefg')

    def test_subdirectory_running_process_blocks_install(self):
        proc = SimpleNamespace(info={'name': 'child.exe', 'exe': str(self.game / 'bin' / 'child.exe')})
        with patch('core.injector.psutil.process_iter', return_value=[proc]):
            self.assertTrue(self.injector.is_game_running(str(self.exe), str(self.game)))

    def test_same_name_unrelated_process_does_not_block(self):
        proc = SimpleNamespace(info={'name': self.exe.name, 'exe': str(self.root / 'other' / self.exe.name)})
        with patch('core.injector.psutil.process_iter', return_value=[proc]):
            self.assertFalse(self.injector.is_game_running(str(self.exe), str(self.game)))

    def test_clean_early_exit_does_not_rollback(self):
        safety = Mock()
        watchdog = CrashWatchdog(str(self.exe), str(self.game), safety, surveillance_duration=1)
        proc = Mock()
        proc.poll.return_value = 0
        with patch('core.safety.psutil.process_iter', return_value=[]):
            watchdog._surveillance_loop(proc, 123)
        safety.rollback.assert_not_called()

    def test_unrelated_reporter_pid_substring_does_not_rollback(self):
        safety = Mock()
        watchdog = CrashWatchdog(str(self.exe), str(self.game), safety, surveillance_duration=1)
        proc = Mock()
        proc.poll.return_value = None
        reporter = Mock(info={'name': 'WerFault.exe'})
        reporter.cmdline.return_value = ['WerFault.exe', '-p', '1234']
        with patch('core.safety.psutil.process_iter', return_value=[reporter]), patch('core.safety.time.time', side_effect=[0, 0, 2]):
            watchdog._surveillance_loop(proc, 123)
        safety.rollback.assert_not_called()

    def test_unknown_hook_rejected(self):
        self.assertFalse(self.apply('unsupported.dll')['success'])

    def test_missing_target_executable_rejected(self):
        self.exe.unlink()
        self.assertFalse(self.apply()['success'])

    def test_revert_preserves_unmanaged_imgui_configuration(self):
        (self.game / 'imgui.ini').write_text('another overlay config')
        self.assertTrue(self.apply()['success'])
        self.injector.revert_injection(str(self.game))
        self.assertTrue((self.game / 'imgui.ini').exists())

    def test_readding_profile_preserves_settings(self):
        from core.library import GameLibrary
        lib = GameLibrary(str(self.root / 'profiles.json'))
        profile = lib.add_game_by_path(str(self.exe))
        lib.update_profile(profile['id'], {'reflex_boost': False, 'custom_scale': 0.7, 'optiscaler_version': 'v0.9.4'})
        profile = lib.add_game_by_path(str(self.exe))
        self.assertEqual(profile.get('custom_scale'), 0.7)

    def test_gui_load_restores_version_and_depth_controls(self):
        from gui.config_tab import ConfigTab
        tab = Mock()
        profile = {'name': 'A', 'engine': 'Generic', 'target_dir': str(self.game),
                   'optiscaler_version': 'v0.9.3', 'invert_depth': True, 'jitter_cancellation': False}
        ConfigTab.load_game(tab, profile)
        self.assertEqual((tab.version_var.set.call_args, tab.invert_depth_var.set.call_args,
                          tab.jitter_var.set.call_args),
                         (unittest.mock.call('v0.9.3'), unittest.mock.call(True), unittest.mock.call(False)))

    def test_gui_watchdog_updates_launched_profile_after_selection_change(self):
        from gui.config_tab import ConfigTab
        tab = Mock()
        original = {'id': 'A', 'name': 'A', 'target_dir': str(self.game), 'target_exe': str(self.exe)}
        tab.current_profile = original
        tab.injector.is_injected.return_value = True
        with patch('gui.config_tab.CrashWatchdog') as cls, patch('gui.config_tab.messagebox.showerror'):
            ConfigTab.launch_and_protect(tab)
            callback = cls.call_args.kwargs['on_crash_detected']
            tab.current_profile = {'id': 'B', 'name': 'B'}
            callback('simulated crash', {'success': True})
        self.assertEqual(tab.library.update_profile.call_args.args[0], 'A')


if __name__ == '__main__':
    unittest.main(verbosity=2)
