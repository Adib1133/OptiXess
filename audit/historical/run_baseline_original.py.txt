"""Run existing tests without overwriting installed DLLs or user profiles.

Network is deliberately offline; the release test exercises its fallback.
GUI construction uses a temporary repository and withdrawn window.
"""
import contextlib
import hashlib
import os
from pathlib import Path
import sys
import tempfile
import unittest
from unittest.mock import patch

ROOT = Path(__file__).resolve().parents[1]
sys.path[:0] = [str(ROOT), str(ROOT / '.audit-deps')]
from core.injector import Injector
from core.library import GameLibrary


def hashes():
    return {str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in (ROOT / 'assets').rglob('*') if p.is_file()}


if __name__ == '__main__':
    before = hashes()
    with tempfile.TemporaryDirectory(prefix='opti_baseline_') as tmp, contextlib.ExitStack() as stack:
        stack.enter_context(patch('urllib.request.urlopen', side_effect=OSError('Isolated offline test')))
        stack.enter_context(patch('gui.main_window.Injector', side_effect=lambda **kw: Injector(os.path.join(tmp, 'assets'))))
        stack.enter_context(patch('gui.main_window.GameLibrary', side_effect=lambda: GameLibrary(os.path.join(tmp, 'profiles.json'))))
        suite = unittest.defaultTestLoader.discover(str(ROOT / 'tests'))
        result = unittest.TextTestRunner(verbosity=2).run(suite)
    assert before == hashes(), 'Installed assets changed during tests'
    print('Installed assets unchanged (SHA-256 verified).')
    sys.exit(not result.wasSuccessful())
